def extract_json(text):
    m = re.search(r'"schema"\s*:\s*"independent-audit/v1"', text)
    if not m: return None
    start = text.rfind('{', 0, m.start()); depth = 0
    for j, ch in enumerate(text[start:]):
        if ch == '{': depth += 1
        elif ch == '}':
            depth -= 1
            if depth == 0:
                try: return json.loads(text[start:start+j+1])
                except Exception: return None
    return None

def grok_report():
    f = RAW/"GROK_AUDIT_RAW.jsonl"
    if not f.exists(): return ""
    texts = []
    for line in f.read_text().splitlines():
        line=line.strip()
        if not line.startswith('{'): continue
        try: o = json.loads(line)
        except Exception: continue
        t = o.get("type")
        if t == "result" and isinstance(o.get("result"), str): texts.append(o["result"])
        elif t in ("text","assistant") :
            if isinstance(o.get("text"), str): texts.append(o["text"])
            c = o.get("message",{}).get("content") if isinstance(o.get("message"),dict) else None
            if isinstance(c, list):
                for part in c:
                    if isinstance(part, dict) and part.get("type")=="text": texts.append(part.get("text",""))
    # prefer the last 'result' if present
    return texts[-1] if texts else ""
