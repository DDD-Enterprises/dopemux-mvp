#!/usr/bin/env python3
"""Cycle-2 proof rebuild for proof/MACRO-DMX-MCP-JUST-WORKS-M1-001 in the governance layout.

Steps (idempotent only when run once on a clean proof dir; refuses if superseded names exist):
  1. git mv every 30b0fdec9 audit artifact under review_bundle/ to a `superseded-30b0fdec9-` name.
  2. Copy the cycle-2 subject, prompts, raw streams, tooling, suite log, review packages and gate
     outputs into review_bundle/ with allowlist-safe names.
  3. Write AUDIT_{AGY,CODEX,GROK}_REPORT.md from the returns; AUDITOR_REPORT.md = governing route.
  4. Rewrite PROOF.json (head, files_changed, slices, routes, validations, audit, embedded_audit)
     keeping `audit_identity` byte-identical; rewrite proof/pr_merge/embedded-audit/pr-1503/PROOF.json.
  5. Rewrite AUDIT_ROUTES.json (new routes + the superseded 30b0fdec9 block) and append a cycle-2
     section to AUDIT_SUMMARY.md.
Usage: build_proof_c2.py <raw_dir> <subject_dir> <suite_log> <gates.json> <meta.json> [--governing agy|grok]
  gates.json: JSON list of {"command","exit_code","bucket","reason"} = the full `validations` list.
  meta.json: {"codereview_status","slices_added":[...],"pr_url_or_blocker","cleanup_status",
              "residual_risks":[...],"unknowns":[...],"summary_md": "...", "fixes_applied_added":[...]}
"""
from __future__ import annotations

import datetime as _dt
import json
import re
import shutil
import subprocess
import sys
from pathlib import Path

W = Path("/Users/hue/code/dopemux-mvp/.worktrees/mcp-just-works-m1")
PACKET = "MACRO-DMX-MCP-JUST-WORKS-M1-001"
P = W / "proof" / PACKET
RB = P / "review_bundle"
PRP = W / "proof/pr_merge/embedded-audit/pr-1503/PROOF.json"
PREV = "30b0fdec9"
START = "5742575c4"
MAIN = "f16e0da2cce256096efeaacb1ade625c466dfb6c"
ALLOWED_RB = re.compile(r"^[A-Za-z0-9][A-Za-z0-9._-]*\.(txt|json|md|diff)$")
ALLOWED_TOP = re.compile(
    r"^(PROOF\.json(\.sig)?|AUDITOR_REPORT\.md|AUDIT_[A-Z0-9_]+\.(md|json)|COMMAND_LOG\.md|HANDOFF\.md|"
    r"THREAT_MODEL\.md|VALIDATION\.json|MODEL_CALL_BUDGET\.md|C\d+_HEAD\.txt|implementation-notes\.md)$"
)
MODELS = {"codex": "gpt-6-astra", "agy": "gemini-4-argon-high", "grok": "grok-4.7"}
TOOLS = {"codex": "codex-cli", "agy": "agy", "grok": "grok-cli"}
SEV = {"low": "LOW", "medium": "MEDIUM", "high": "HIGH", "critical": "BLOCKING", "info": "INFO"}


def git(*args: str, check: bool = True) -> str:
    r = subprocess.run(["git", "-C", str(W), *args], capture_output=True, text=True)
    if check and r.returncode != 0:
        raise SystemExit(f"git {' '.join(args)} failed: {r.stderr.strip()}")
    return r.stdout.strip()


def extract_json(text: str):
    """The independent-audit/v1 verdict object: the LAST such object in ``text``.

    ``json.JSONDecoder.raw_decode`` from each candidate '{' handles braces inside
    strings (``${VAR}``, ``{{.ID}}`` in evidence) that a brace counter mis-balances.
    """
    dec = json.JSONDecoder()
    found = None
    for m in re.finditer(r'"schema"\s*:\s*"independent-audit/v1"', text):
        start = text.rfind("{", 0, m.start())
        candidate = text[start:]
        for attempt in (candidate, _repair_json(candidate)):
            try:
                obj, _ = dec.raw_decode(attempt)
            except Exception:  # noqa: BLE001
                continue
            if isinstance(obj, dict) and obj.get("schema") == "independent-audit/v1":
                found = obj
                if attempt is not candidate:
                    JSON_REPAIRS.append(
                        "verdict JSON contained a non-JSON string expression "
                        "'\"<hex>\"[0:0] + \"<hex>\"' in file_hashes (model output glitch); "
                        "repaired to the second literal, which is the expression's value; "
                        "the raw return is kept verbatim in review_bundle/"
                    )
                break
    return found


JSON_REPAIRS: list = []
_CONCAT_GLITCH = re.compile(r'"[0-9a-f]{64}"\[0:0\]\s*\+\s*("[0-9a-f]{64}")')


def _repair_json(text: str) -> str:
    """Targeted repair of one observed model glitch; anything else stays unparsed."""
    return _CONCAT_GLITCH.sub(r"\1", text)


def agy_text(raw: Path) -> str:
    f = raw / "agy.json"
    if not f.exists():
        return ""
    try:
        o = json.loads(f.read_text(encoding="utf-8", errors="replace"))
    except Exception:  # noqa: BLE001
        return ""
    return o.get("response") or o.get("result") or ""


def grok_text(raw: Path) -> str:
    f = raw / "grok.jsonl"
    if not f.exists():
        return ""
    texts = []
    for line in f.read_text(encoding="utf-8", errors="replace").splitlines():
        line = line.strip()
        if not line.startswith("{"):
            continue
        try:
            o = json.loads(line)
        except Exception:  # noqa: BLE001
            continue
        t = o.get("type")
        if t == "result" and isinstance(o.get("result"), str):
            texts.append(o["result"])
        elif t in ("text", "assistant"):
            if isinstance(o.get("text"), str):
                texts.append(o["text"])
            msg = o.get("message")
            c = msg.get("content") if isinstance(msg, dict) else None
            if isinstance(c, list):
                for part in c:
                    if isinstance(part, dict) and part.get("type") == "text":
                        texts.append(part.get("text", ""))
    return texts[-1] if texts else ""


def codex_text(raw: Path) -> str:
    f = raw / "codex-report.md"
    return f.read_text(encoding="utf-8", errors="replace") if f.exists() else ""


def exit_code(raw: Path, name: str):
    f = raw / f"{name}.stderr"
    if not f.exists():
        return None
    m = re.findall(r"exit=(\d+)", f.read_text(encoding="utf-8", errors="replace"))
    return int(m[-1]) if m else None


def safe_copy(src: Path, dst_name: str) -> None:
    if not ALLOWED_RB.match(dst_name):
        raise SystemExit(f"unsafe review_bundle name {dst_name!r}")
    # governance text check flags non-UTF-8 evidence (info level): re-encode with replacement
    text = src.read_bytes().decode("utf-8", errors="replace")
    (RB / dst_name).write_text(text, encoding="utf-8")


def main() -> None:
    args = [a for a in sys.argv[1:] if not a.startswith("--")]
    governing = "agy"
    if "--governing" in sys.argv:
        governing = sys.argv[sys.argv.index("--governing") + 1]
    raw, subject, suite_log, gates_path, meta_path = (Path(a) for a in args[:5])
    head = git("rev-parse", "HEAD")
    h8 = head[:8]
    if git("status", "--porcelain", "--", "proof"):
        raise SystemExit("refusing: proof/ has uncommitted changes")
    if any(p.name.startswith(f"superseded-{PREV}-") for p in RB.iterdir()):
        raise SystemExit(f"refusing: superseded-{PREV}- names already exist (already rebuilt?)")
    base_tree = (subject / "BASE_TREE").read_text().strip()
    nonce = (subject / "NONCE").read_text().strip()
    gates = json.loads(gates_path.read_text(encoding="utf-8"))
    meta = json.loads(meta_path.read_text(encoding="utf-8"))

    # 1. supersede the 30b0fdec9 artifacts
    keep = {"acceptance-log.txt", "owner-cleanup-m1-acceptance.sh.txt", "grok-disallowed.txt", f"test-fast-{PREV}.txt"}
    for f in sorted(RB.iterdir()):
        if f.name in keep or f.name.startswith("superseded-"):
            continue
        new = f"superseded-{PREV}-{f.name}"
        if not ALLOWED_RB.match(new):
            raise SystemExit(f"unsafe name {new}")
        git("mv", f"proof/{PACKET}/review_bundle/{f.name}", f"proof/{PACKET}/review_bundle/{new}")
    for rep in ("AUDIT_AGY_REPORT.md", "AUDIT_CODEX_REPORT.md", "AUDIT_GROK_REPORT.md"):
        if (P / rep).exists():
            git("mv", f"proof/{PACKET}/{rep}", f"proof/{PACKET}/review_bundle/superseded-{PREV}-{rep}")

    # 2. cycle-2 evidence
    copies = {
        subject / "AUDIT_PROMPT.codex.txt": "audit-prompt-codex.txt",
        subject / "AUDIT_PROMPT.agy.txt": "audit-prompt-agy.txt",
        subject / "AUDIT_PROMPT.grok.txt": "audit-prompt-grok.txt",
        subject / "AUDIT_PROMPT.grok.full.txt": "audit-prompt-grok-full.txt",
        subject / "COMMITS.txt": "subject-commits.txt",
        subject / "FILES.sha256": "subject-files-sha256.txt",
        subject / "subject.diff": "subject.diff",
        subject / "subject-src.diff": "subject-src.diff",
        subject / "VALIDATIONS.md": "subject-validations.md",
        subject / "LEDGER_SUMMARY.md": "subject-ledger-summary.md",
        subject / "BASE_TREE": "subject-base-tree.txt",
        raw / "codex.jsonl": "raw-codex.txt",
        raw / "codex.stderr": "raw-codex-stderr.txt",
        raw / "agy.json": "raw-agy.json",
        raw / "agy.stderr": "raw-agy-stderr.txt",
        raw / "grok.jsonl": "raw-grok.txt",
        raw / "grok.stderr": "raw-grok-stderr.txt",
        raw / "started": "raw-started.txt",
        suite_log: f"test-fast-{h8}.txt",
        Path(__file__): "build_proof_c2.py.txt",
        Path(__file__).with_name("cycle2-freeze-and-audit.sh"): "cycle2-freeze-and-audit.sh.txt",
        Path(__file__).with_name("AUDIT_PROMPT.c2.template.txt"): "audit-prompt-c2-template.txt",
        gates_path: f"cycle2-gates-{h8}.json",
    }
    optional = {
        Path(__file__).with_name("cycle2-review-A.diff"): "cycle2-review-A.diff",
        Path(__file__).with_name("cycle2-review-B.diff"): "cycle2-review-B.diff",
        Path(__file__).with_name("cycle2-review-final.diff"): "cycle2-review-final.diff",
        Path(__file__).with_name("cycle2-bot-reviews.json"): "cycle2-bot-reviews.json",
        Path(__file__).with_name("cycle2-live-smoke.txt"): f"cycle2-live-smoke-{h8}.txt",
        Path(__file__).with_name("cycle2-task-A-brief.md"): "cycle2-task-A-brief.md",
        Path(__file__).with_name("cycle2-task-B-brief.md"): "cycle2-task-B-brief.md",
        Path(__file__).with_name("cycle2-task-A-report.md"): "cycle2-task-A-report.md",
        Path(__file__).with_name("cycle2-task-B-report.md"): "cycle2-task-B-report.md",
    }
    for src, dst in copies.items():
        if not src.is_file():
            raise SystemExit(f"missing evidence source: {src}")
        safe_copy(src, dst)
    for src, dst in optional.items():
        if src.is_file():
            safe_copy(src, dst)

    # 3. reports + verdicts
    texts = {"codex": codex_text(raw), "agy": agy_text(raw), "grok": grok_text(raw)}
    verdicts = {}
    routes = []
    for r in ("codex", "agy", "grok"):
        text = texts[r]
        if text:
            (P / f"AUDIT_{r.upper()}_REPORT.md").write_text(text, encoding="utf-8")
        v = extract_json(text) if text else None
        verdicts[r] = v
        marker_ok = (f"END_OF_PROMPT {nonce}-{h8}" in text) or (f"END_OF_SUBJECT {nonce}-{h8}" in text)
        routes.append(
            {
                "workstream": f"independent L3 audit route — {r} (cycle 2 at {h8})",
                "runner": r,
                "model_requested": MODELS[r],
                "model_observed": "UNKNOWN",
                "effort": "high" if r == "grok" else "UNKNOWN",
                "skills": ["independent-audit"],
                "status": (v or {}).get("status", "NOT_RUN" if not text else "UNKNOWN"),
                "exit_code": exit_code(raw, r),
                "marker_echoed": marker_ok,
                "report_path": f"proof/{PACKET}/AUDIT_{r.upper()}_REPORT.md",
                "raw_path": f"proof/{PACKET}/review_bundle/raw-{r}*",
                "verdict_json_repair": (JSON_REPAIRS[-1] if (r == "agy" and JSON_REPAIRS) else None),
                "findings": [
                    {"id": f.get("id"), "severity": f.get("severity"), "location": f.get("location"), "summary": f.get("summary")}
                    for f in (v or {}).get("findings", [])
                ],
            }
        )
    gov = verdicts.get(governing) or {}
    if not gov:
        raise SystemExit(f"governing route {governing} returned no verdict JSON")
    shutil.copyfile(P / f"AUDIT_{governing.upper()}_REPORT.md", P / "AUDITOR_REPORT.md")

    # 4. PROOF.json
    proof = json.loads((P / "PROOF.json").read_text(encoding="utf-8"))
    identity_before = json.dumps(proof["audit_identity"], sort_keys=True)
    if governing != "agy":
        raise SystemExit("governing route other than agy changes audit_identity; refuse (owner decision)")
    proof["head_sha"] = head
    proof["commit_sha"] = head
    proof["files_changed"] = git("diff", "--name-only", base_tree, head, "--", ".", ":(exclude)proof").splitlines()
    proof["slices"] = list(proof.get("slices", [])) + list(meta.get("slices_added", []))
    kept_routes = [r for r in proof["routes"] if not str(r.get("workstream", "")).startswith("independent L3 audit route")]
    proof["routes"] = kept_routes + list(meta.get("routes_added", [])) + routes
    proof["validations"] = gates
    proof["codereview_status"] = meta["codereview_status"]
    proof["audit"] = {
        "required": True,
        "status": {r: (verdicts[r] or {}).get("status", "NOT_RUN/UNKNOWN") for r in ("codex", "agy", "grok")},
        "route": (
            f"three routes at the frozen head {h8} (codex gpt-6-astra, agy gemini-4-argon-high, grok grok-4.7); "
            f"embedded_audit below carries the {governing} record (schema-valid); the 30b0fdec9 round is superseded "
            f"(review_bundle/superseded-{PREV}-*)"
        ),
    }
    proof["pr_url_or_blocker"] = meta["pr_url_or_blocker"]
    proof["residual_risks"] = list(meta.get("residual_risks", []))
    proof["unknowns"] = list(meta.get("unknowns", []))
    proof["cleanup_status"] = meta["cleanup_status"]
    ea_old = proof["embedded_audit"]
    ilc = gov.get("instruction_like_content", []) or []
    proof["embedded_audit"] = {
        "required": True,
        "status": gov.get("status", "NEEDS_SUPERVISOR"),
        "auditor_tool": "agy",
        "auditor_model": "gemini",
        "invocation": (
            "agy --model gemini-4-argon-high --output-format json --print-timeout 25m --add-dir <sealed subject> "
            "--print=<AUDIT_PROMPT.agy.full.txt> (exact requested model gemini-4-argon-high; the schema enum lacks it, hence 'gemini')"
        ),
        "exit_code": exit_code(raw, "agy"),
        "report_path": f"proof/{PACKET}/AUDITOR_REPORT.md",
        "findings": [
            {
                "id": f.get("id", "F?"),
                "severity": SEV.get(str(f.get("severity", "")).lower(), "INFO"),
                "title": (f.get("summary") or "")[:120] or "finding",
                "status": "ACCEPTED_RISK" if str(f.get("severity", "")).lower() == "low" else "OPEN",
                "body": f"{f.get('location', '')}: {f.get('summary', '')} — evidence: {f.get('evidence', '')}",
            }
            for f in gov.get("findings", [])
        ],
        "fixes_applied": list(ea_old.get("fixes_applied", [])) + list(meta.get("fixes_applied_added", [])),
        "remaining_risks": list(gov.get("remaining_risks", []))
        + [f"{r} route at {h8} = {(verdicts[r] or {}).get('status', 'UNKNOWN')}" for r in ("codex", "grok")],
        "skip_reason": None,
        "instruction_like_content": {"detected": bool(ilc), "match_count": len(ilc), "truncated": False, "matches": []},
    }
    proof["generated_at"] = _dt.datetime.now(_dt.timezone.utc).isoformat()
    assert json.dumps(proof["audit_identity"], sort_keys=True) == identity_before
    (P / "PROOF.json").write_text(json.dumps(proof, indent=2) + "\n", encoding="utf-8")

    prp = json.loads(PRP.read_text(encoding="utf-8"))
    prp["head_sha"] = head
    prp["generated_at"] = proof["generated_at"]
    prp["embedded_audit"] = proof["embedded_audit"]
    prp["audit_identity"] = proof["audit_identity"]
    PRP.write_text(json.dumps(prp, indent=2) + "\n", encoding="utf-8")

    # 5. AUDIT_ROUTES.json + AUDIT_SUMMARY.md
    old_routes = json.loads((P / "AUDIT_ROUTES.json").read_text(encoding="utf-8"))
    (P / "AUDIT_ROUTES.json").write_text(
        json.dumps(
            {
                "head_sha": head,
                "nonce": nonce,
                "base_tree": base_tree,
                "base_rule": f"merge-tree({START}, {MAIN}); diff excludes proof/",
                "routes": routes,
                "verdicts": verdicts,
                "superseded": [
                    {k: old_routes.get(k) for k in ("head_sha", "nonce", "routes", "verdicts")},
                    *old_routes.get("superseded", []),
                ],
            },
            indent=2,
        )
        + "\n",
        encoding="utf-8",
    )
    table = "\n".join(
        f"| {r} | {MODELS[r]} | {(verdicts[r] or {}).get('status', 'NOT_RUN/UNKNOWN')} | "
        + ("; ".join(f"{f.get('id')} {f.get('severity')} {f.get('location')}" for f in (verdicts[r] or {}).get("findings", [])) or "none")
        + " |"
        for r in ("agy", "grok", "codex")
    )
    summ = P / "AUDIT_SUMMARY.md"
    summ.write_text(
        summ.read_text(encoding="utf-8").rstrip("\n")
        + f"\n\n## Cycle 2 ({_dt.date.today().isoformat()}) — forced fix + re-audit at {head}\n\n"
        + meta["summary_md"].rstrip("\n")
        + "\n\n### Audit (three routes at "
        + h8
        + "; observed identity UNKNOWN for all)\n\n| route | requested model | status | findings |\n|---|---|---|---|\n"
        + table
        + "\n",
        encoding="utf-8",
    )

    for f in RB.iterdir():
        assert ALLOWED_RB.match(f.name), f.name
    for f in P.iterdir():
        if f.is_file():
            assert ALLOWED_TOP.match(f.name), f.name
    subprocess.run(["git", "-C", str(W), "add", "-f", "--", f"proof/{PACKET}", str(PRP.relative_to(W))], check=True)
    print("rebuilt", {r: routes[i]["status"] for i, r in enumerate(("codex", "agy", "grok"))}, "head", head)
    print("review_bundle files:", len(list(RB.iterdir())))


if __name__ == "__main__":
    main()
