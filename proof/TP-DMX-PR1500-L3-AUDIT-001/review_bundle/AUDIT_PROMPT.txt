INTENT
PR 1498 added `dopemux auth`, which issues, rotates and revokes bearer tokens. Tokens live in the macOS keychain, and the memory servers (dope-memory, ConPort) verify a SHA-256 hash from a shared JSON file. A review of that PR found two defects. No token has been issued yet.
(a) Race. `issue` and `rotate` loaded the hash file, called the keychain, then rewrote the whole file from the copy they had loaded. A `revoke` that finished in between was undone, so the revoked token verified again. The fix adds `hashfile.exclusive_lock()`, built on the third-party `filelock` package, and holds it across the load, the keychain call, the hash write and the metadata write in `issue`, `rotate` and `revoke`. Preconditions are checked inside it. A busy lock raises HashFileError after 30 seconds. Dry runs and `list` take no lock. The servers only read the file, and writes are atomic renames.
(b) Trailing newline. `$` also matches before a final newline, so `.match` accepted `runner:codex\n`. A principal is interpolated into a line sent to `security -i` on stdin. The fix changes six patterns to end in `\Z`: PRINCIPAL_RE, _WRITABLE_TOKEN_RE and _INTERNAL_PRINCIPAL_RE in the CLI package, and _PRINCIPAL_RE and _SHA256_HEX_RE in both servers.

     Returns ``(workspace_id, instance_id)``.  An omitted, empty or blank

 class AuthGate:
     """Verifies bearer tokens and counts unverified callers."""
 
     def __init__(self, mode: str, auth_dir: Optional[Path | str] = None):
         if mode not in AUTH_MODES:
             raise ValueError(f"invalid auth mode {mode!r}; use one of {', '.join(AUTH_MODES)}")
         self.mode = mode
         self._hash_path = Path(auth_dir) / HASH_FILE_NAME if auth_dir else None
         self._lock = threading.Lock()
         self._stamp: Optional[tuple[int, int]] = None
         self._hashes: dict[str, str] = {}
         self._status = "unconfigured" if self._hash_path is None else "missing"
         self._unverified: dict[str, int] = {}
 
     @classmethod
     def from_env(cls, env: Optional[Mapping[str, str]] = None) -> "AuthGate":
         source = os.environ if env is None else env
         return cls(load_auth_mode(source), (source.get(ENV_AUTH_DIR) or "").strip() or None)
 
     # -- hash file ---------------------------------------------------------
 
     def _refresh(self) -> tuple[dict[str, str], str]:
         if self._hash_path is None:
             return {}, "unconfigured"
         with self._lock:
             try:
                 stat = self._hash_path.stat()
             except OSError:
                 self._stamp, self._hashes, self._status = None, {}, "missing"
                 return self._hashes, self._status
             stamp = (stat.st_mtime_ns, stat.st_size)
             if stamp != self._stamp:
                 self._stamp = stamp
                 self._hashes, self._status = self._read_hash_file()
             return self._hashes, self._status
 
     def _read_hash_file(self) -> tuple[dict[str, str], str]:
         assert self._hash_path is not None
         try:
             data = json.loads(self._hash_path.read_text(encoding="utf-8"))
         except (OSError, ValueError):
             logger.error("ConPort auth hash file is unreadable or not JSON")
             return {}, "invalid"
         if not isinstance(data, dict):
             logger.error("ConPort auth hash file is not a JSON object")
             return {}, "invalid"
         hashes: dict[str, str] = {}
         skipped = 0
         for digest, principal in data.items():
             if (
                 isinstance(digest, str)
                 and isinstance(principal, str)
                 and _SHA256_HEX_RE.match(digest)
                 and _PRINCIPAL_RE.match(principal)
             ):
                 hashes[digest] = principal
             else:
                 skipped += 1
         if skipped:
             logger.warning("ConPort auth hash file: skipped %d malformed entries", skipped)
         return hashes, "ok"
 
     # -- verification ----------------------------------------------------------
 
     def authenticate(self, authorization: Optional[str]) -> Identity:
         """Return the verified principal for a header value, else ``unverified``."""
         token = extract_bearer(authorization)
         if token is None:
             return UNVERIFIED
         hashes, _ = self._refresh()
         digest = hashlib.sha256(token.encode("utf-8")).hexdigest()
         matched: Optional[str] = None
         for stored, principal in hashes.items():
             # No early exit: every stored hash is compared.
             if hmac.compare_digest(digest, stored):
                 matched = principal
         return Identity(matched, True) if matched else UNVERIFIED
 
     def record_unverified(self, caller: str) -> None:
         with self._lock:
             if caller in self._unverified:
                 self._unverified[caller] += 1
                 return
             if len(self._unverified) < MAX_TRACKED_CALLERS:
                 self._unverified[caller] = 1
                 logger.warning("unverified ConPort write from caller=%s", caller)
             else:
                 self._unverified["other"] = self._unverified.get("other", 0) + 1
 
     def authorize_write(self, authorization: Optional[str], caller: str) -> Identity:
         """Apply the mode to a write; return the identity to store with it."""
         if self.mode == "off":
             return UNVERIFIED
         identity = self.authenticate(authorization)
         if identity.verified:
             return identity
         self.record_unverified(caller)
         if self.mode == "enforce":
             raise Unauthorized(
                 "unauthorized (401): a valid bearer token is required for writes "
                 "while the auth mode is 'enforce'"
             )
         return identity
 
     def health_fields(self) -> dict[str, Any]:
         """Fields the health route reports (read by the generator gate and doctor)."""
         _, status = self._refresh()

         return {

             "identity_required_writes": True,
             "auth_hash_file": status,

         }

     gate: AuthGate,
     authorization: Optional[str],
     caller: str,
     *,
     fallback_token: Any = None,
 ) -> dict[str, str]:
     """Headers an in-image proxy sends to the REST port for one write.
 
     - the caller's own bearer token is forwarded as sent (REST verifies it again);
     - with no caller token, in ``log`` mode only, ``fallback_token`` (the
       ``internal:bridge-proxy`` token, when one is available; a string or a
       callable returning one, called only at this point) is sent instead;
     - an unverified caller is labelled in ``UNVERIFIED_CALLER_HEADER`` so REST
       can count it even when the proxy's own token made the request verified;

     - ``off`` mode forwards only what the caller sent.
 
     The token never reaches a log line.
     """

         if token:
             headers["Authorization"] = f"Bearer {token}"
         return headers

 class AuthGate:
     """Verifies bearer tokens and counts unverified callers."""
 
     def __init__(self, mode: str, auth_dir: Optional[Path | str] = None):
         if mode not in AUTH_MODES:
             raise ValueError(f"invalid auth mode {mode!r}; use one of {', '.join(AUTH_MODES)}")
         self.mode = mode
         self._hash_path = Path(auth_dir) / HASH_FILE_NAME if auth_dir else None
         self._lock = threading.Lock()
         self._stamp: Optional[tuple[int, int]] = None
         self._hashes: dict[str, str] = {}
         self._status = "unconfigured" if self._hash_path is None else "missing"
         self._unverified: dict[str, int] = {}
 
     @classmethod
     def from_env(cls, env: Optional[Mapping[str, str]] = None) -> "AuthGate":
         source = os.environ if env is None else env
         return cls(load_auth_mode(source), (source.get(ENV_AUTH_DIR) or "").strip() or None)
 
     # -- hash file ---------------------------------------------------------
 
     def _refresh(self) -> tuple[dict[str, str], str]:
         if self._hash_path is None:
             return {}, "unconfigured"
         with self._lock:
             try:
                 stat = self._hash_path.stat()
             except OSError:
                 self._stamp, self._hashes, self._status = None, {}, "missing"
                 return self._hashes, self._status
             stamp = (stat.st_mtime_ns, stat.st_size)
             if stamp != self._stamp:
                 self._stamp = stamp
                 self._hashes, self._status = self._read_hash_file()
             return self._hashes, self._status
 
     def _read_hash_file(self) -> tuple[dict[str, str], str]:
         assert self._hash_path is not None
         try:
             data = json.loads(self._hash_path.read_text(encoding="utf-8"))
         except (OSError, ValueError):
             logger.error("dope-memory auth hash file is unreadable or not JSON")
             return {}, "invalid"
         if not isinstance(data, dict):
             logger.error("dope-memory auth hash file is not a JSON object")
             return {}, "invalid"
         hashes: dict[str, str] = {}
         skipped = 0
         for digest, principal in data.items():
             if (
                 isinstance(digest, str)
                 and isinstance(principal, str)
                 and _SHA256_HEX_RE.match(digest)
                 and _PRINCIPAL_RE.match(principal)
             ):
                 hashes[digest] = principal
             else:
                 skipped += 1
         if skipped:
             logger.warning("dope-memory auth hash file: skipped %d malformed entries", skipped)
         return hashes, "ok"
 
     # -- verification ----------------------------------------------------------
 
     def authenticate(self, authorization: Optional[str]) -> Identity:
         """Return the verified principal for a header value, else ``unverified``."""
         token = _extract_bearer(authorization)
         if token is None:
             return UNVERIFIED
         hashes, _ = self._refresh()
         digest = hashlib.sha256(token.encode("utf-8")).hexdigest()
         matched: Optional[str] = None
         for stored, principal in hashes.items():
             # No early exit: every stored hash is compared.
             if hmac.compare_digest(digest, stored):
                 matched = principal
         return Identity(matched, True) if matched else UNVERIFIED
 
     def record_unverified(self, caller: str) -> None:
         with self._lock:
             if caller in self._unverified:
                 self._unverified[caller] += 1
                 return
             if len(self._unverified) < MAX_TRACKED_CALLERS:
                 self._unverified[caller] = 1
                 logger.warning("unverified dope-memory write from caller=%s", caller)
             else:
                 self._unverified["other"] = self._unverified.get("other", 0) + 1
 
     def authorize_write(self, authorization: Optional[str], caller: str) -> Identity:
         """Apply the mode to a write; return the identity to store with it."""
         if self.mode == "off":
             return UNVERIFIED
         identity = self.authenticate(authorization)
         if identity.verified:
             return identity
         self.record_unverified(caller)
         if self.mode == "enforce":
             raise Unauthorized(
                 "unauthorized (401): a valid bearer token is required for writes "
                 "while the auth mode is 'enforce'"
             )
         return identity
 
     def health_fields(self) -> dict[str, Any]:
         """Fields the health route reports (read by the generator gate and doctor)."""
         _, status = self._refresh()
         with self._lock:
             counts = dict(self._unverified)
         # identity_required_writes stays True: it means writes are bound to this
         # server's workspace (a mismatched workspace is rejected), not that the
         # caller must send the ids (omitted ids are filled from the binding).
         return {
             "auth_mode": self.mode,
             "identity_required_writes": True,
             "auth_hash_file": status,
             "auth_unverified_calls": counts,
         }

 def _ensure_schema_initialized(
     conn: sqlite3.Connection,
     repo_root: Path,
     schema_path: Path,
     migrations_dir: Path,
     ledger_path: Path,
 ) -> None:
     ledger_key = str(ledger_path.resolve())
     if ledger_key in _SCHEMA_READY_LEDGER_PATHS:
         return
 
     with _SCHEMA_INIT_LOCK:
         if ledger_key in _SCHEMA_READY_LEDGER_PATHS:
             return
         migration_applier = _load_wma_sqlite_migration_applier(repo_root)
         try:
             migration_applier(
                 conn,
                 schema_path=schema_path,
                 migrations_dir=migrations_dir,
                 logger=LOGGER,
             )
         except Exception as exc:
             raise CaptureError(
                 f"Failed to initialize/migrate chronicle schema for {ledger_path}: {exc}"
             ) from exc
         _SCHEMA_READY_LEDGER_PATHS.add(ledger_key)

 def _deterministic_event_id(
     *,
     event_type: str,
     session_id: Optional[str],
     ts_utc: str,
     payload: dict[str, Any],
 ) -> str:
     """Generate deterministic event_id from semantic content only.
     
     Per Packet D §3.3: event_id is content-addressed, excluding adapter-specific
     metadata (source, project_id) to enable cross-adapter convergence.
     
     Fingerprint formula: event_type | session_id_or_empty | ts_bucket | stable_json(payload)
     
     Note: session_id_or_empty MUST be empty string ("") when session_id is None,
     not a sentinel value, to keep event_id stable across adapters and environments.
     """
     # Bucket to second precision for retry-idempotency.
     ts_bucket = ts_utc[:19]
     # Normalize session_id to empty string if None (per Packet D clarification)
     session_id_normalized = session_id or ""
     fingerprint = "|".join(
         [
             event_type,
             session_id_normalized,
             ts_bucket,
             _stable_json(payload),
         ]
     )
     return hashlib.sha256(fingerprint.encode("utf-8")).hexdigest()
 
 
 def _emit_to_event_stream(event: dict[str, Any]) -> None:
     """Best-effort Redis stream fan-out for downstream consumers."""
     try:
         import redis  # type: ignore
     except Exception:
         LOGGER.debug("redis package unavailable; skipping event stream emit")
         return
 
     redis_url = os.getenv("REDIS_URL", "redis://localhost:6379")
     stream_name = os.getenv("DOPE_MEMORY_INPUT_STREAM", "activity.events.v1")
     redis_password = os.getenv("REDIS_PASSWORD")
 
     try:
         # Bounded socket timeouts so an unreachable / blackholed REDIS_URL
         # cannot stall a synchronous caller (e.g. a fail-open Claude hook) on
         # the OS TCP timeout. Mirrors native_hooks._open_activity_redis_client. The
         # ledger write has already happened before this fan-out, so a slow or
         # dead Redis degrades to a fast best-effort skip, never a hang.
         client = redis.Redis.from_url(
             redis_url,
             password=redis_password,
             decode_responses=True,
             socket_connect_timeout=1,
             socket_timeout=2,
         )
         envelope = {
             "id": event["id"],
             "ts": event["ts_utc"],
             "workspace_id": event["workspace_id"],
             "instance_id": event["instance_id"],
             "session_id": event.get("session_id") or "",
             "type": event["event_type"],
             "source": event["source"],
             "data": _stable_json(event["payload"]),
         }
         if event.get("write_principal"):
             envelope["write_principal"] = event["write_principal"]
             envelope["write_principal_self_asserted"] = "1"
         client.xadd(stream_name, envelope)
     except Exception as exc:  # pragma: no cover - best effort
         LOGGER.debug("event stream emit failed: %s", exc)
 
 
 def emit_promotable_capture_event(
     event_type: str,
     payload: dict[str, Any],
     *,
     source: str,
     mode: str = CAPTURE_MODE_AUTO,
     repo_root: Optional[Path] = None,
     emit_event_bus: Optional[bool] = False,

 ) -> CaptureResult:

         "source": source,
         "payload": payload,

     return emit_capture_event(
         event,
         mode=mode,
         repo_root=repo_root,
         emit_event_bus=emit_event_bus,
         lane=lane,
         principal=principal,
     )

 def emit_capture_event(
     event: dict[str, Any],
     *,
     mode: str = CAPTURE_MODE_AUTO,
     repo_root: Optional[Path] = None,
     emit_event_bus: Optional[bool] = None,
     lane: Optional[str] = None,
     principal: Optional[str] = None,
 ) -> CaptureResult:
     """Redact + write raw activity event to canonical per-project ledger.
 
     ``principal`` is the self-asserted ``internal:<name>`` recorded on the row
     (default ``internal:cli``); native_hooks passes ``internal:hooks``. It is
     never read from ``event``, so a payload cannot choose its own identity.
     """
     write_principal = _resolve_write_principal(principal)
     root = repo_root.resolve() if repo_root else resolve_repo_root_strict()
     selected_mode = resolve_capture_mode(mode, repo_root=root)
     source = str(event.get("source") or _default_source_for_mode(selected_mode))
 
     event_type = str(
         event.get("event_type")
         or event.get("type")
         or ""
     ).strip()
     if not event_type:
         raise CaptureError("event_type is required")
 
     # Lane policy enforcement (CLI-INT-005)
     from .lane_policy import LanePolicy
 
     # Detect lane only from explicit inputs for deterministic CLI/plugin behavior.
     lane_id = lane or event.get("lane")
 
     # Enforce lane policy only when a lane is explicitly provided.
     if lane_id:
         policy = LanePolicy(root)
         if not policy.should_capture(lane_id, event_type):
             # Policy denied - return non-inserted result
             resolved_ledger_path = _resolve_ledger_path(root)
             return CaptureResult(
                 event_id="",
                 inserted=False,
                 ledger_path=resolved_ledger_path,
                 repo_root=root,
                 mode=selected_mode,
                 source=source,
                 event_type=event_type,
             )
 
     payload = _normalize_payload(event.get("payload", event.get("data")))
     ts_utc = str(event.get("ts_utc") or event.get("ts") or datetime.now(timezone.utc).isoformat())
     session_id = event.get("session_id")
     instance_id = str(event.get("instance_id") or os.getenv("DOPEMUX_INSTANCE_ID", "A"))
 
     project_id = str(root)
     workspace_id = str(event.get("workspace_id") or project_id)
 
     redactor = _load_wma_redactor(root)
     redacted_payload = redactor.redact_payload(payload)
 
     # Generate event_id BEFORE injecting project_id (Packet D §3.3)
     # project_id is machine-specific and must not affect event identity
     event_id = str(
         event.get("event_id")
         or event.get("id")
         or _deterministic_event_id(
             event_type=event_type,
             session_id=session_id,
             ts_utc=ts_utc,
             payload=redacted_payload,
         )
     )
 
     # Inject project_id AFTER event_id generation for storage/routing only
     redacted_payload["project_id"] = project_id
 
     ledger_path = _resolve_ledger_path(root)
     ledger_path.parent.mkdir(parents=True, exist_ok=True)
     schema_path = _resolve_wma_schema_path(root)
     migrations_dir = _resolve_wma_migrations_dir(root)
 
     conn = sqlite3.connect(str(ledger_path))
     try:
         conn.execute("PRAGMA foreign_keys = ON")
         conn.execute("PRAGMA journal_mode = WAL")
         _ensure_schema_initialized(
             conn,
             root,
             schema_path,
             migrations_dir,
             ledger_path,
         )
         cur = conn.execute(
             """
             INSERT OR IGNORE INTO raw_activity_events (
                 id, workspace_id, instance_id, session_id,
                 ts_utc, event_type, source,
                 payload_json, redaction_level, ttl_days, created_at_utc,
                 write_principal, write_principal_self_asserted
             ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, 1)
             """,
             (
                 event_id,
                 workspace_id,
                 instance_id,
                 session_id,
                 ts_utc,
                 event_type,
                 source,
                 json.dumps(redacted_payload, ensure_ascii=True),
                 "strict",
                 int(event.get("ttl_days", 7)),
                 datetime.now(timezone.utc).isoformat(),
                 write_principal,
             ),
         )
         conn.commit()
         inserted = cur.rowcount > 0
     finally:
         conn.close()
 
     should_emit_event_bus = emit_event_bus
     if should_emit_event_bus is None:
         should_emit_event_bus = (
             os.getenv("DOPEMUX_CAPTURE_EMIT_EVENTBUS", "false").strip().lower()
             in {"1", "true", "yes", "on"}
         )
 
     if should_emit_event_bus:
         _emit_to_event_stream(
             {
                 "id": event_id,
                 "workspace_id": workspace_id,
                 "instance_id": instance_id,
                 "session_id": session_id,
                 "event_type": event_type,
                 "source": source,
                 "payload": redacted_payload,
                 "ts_utc": ts_utc,
                 "write_principal": write_principal,
             }
         )
 
     return CaptureResult(
         event_id=event_id,
         inserted=inserted,
         ledger_path=ledger_path,
         repo_root=root,
         mode=selected_mode,
         source=source,
         event_type=event_type,
     )

 class _Recorder:
     def __init__(self):
         self.calls = 0
 
     async def __call__(self, scope, receive, send):
         self.calls += 1
         if scope["type"] != "http":
             return
         body = b""
         while True:
             message = await receive()
             body += message.get("body", b"")
             if not message.get("more_body"):
                 break
         await send({"type": "http.response.start", "status": 200, "headers": []})
         await send({"type": "http.response.body", "body": b"seen:" + body})

+"""Overlapping ``dopemux auth --apply`` commands serialize on one lock.
+
+PR 1498's review found the window: a command read the hash file, waited on the
+keychain, then wrote back what it had read, so a revoke that finished in between
+was undone and the revoked token verified again.
+
+``run_overlapping`` holds the first command in that window, between its keychain
+write and its hash write, and runs the second command there. The click callbacks
+are called directly because ``CliRunner`` swaps ``sys.stdout`` process-wide and
+cannot run in two threads at once.
+"""
+
+from __future__ import annotations
+
+import json
+import os
+import stat
+import subprocess
+import sys
+import threading
+from collections.abc import Callable
+
+import click
+import pytest
+
+from dopemux.auth import hash_token, hashfile
+from dopemux.commands import auth_commands
+from dopemux.commands.auth_commands import issue, list_principals, revoke, rotate
+
+# How long the first command stays in the window. Without the lock the second
+# command finishes well inside this; with it the second command waits it out.
+# A slow machine can only hide the bug, never fail a correct build.
+WINDOW_SECONDS = 0.5
+JOIN_SECONDS = 30.0
+
+
+def run_overlapping(
+    env, first: Callable[[], None], second: Callable[[], None]
+) -> dict[str, click.ClickException]:
+    """Run ``second`` inside ``first``'s window; return each one's refusal, if any."""
+    real_store = auth_commands.store_token
+    in_window = threading.Event()
+    second_finished = threading.Event()
+    refusals: dict[str, click.ClickException] = {}
+
+    def store_then_hold(principal: str, token: str) -> None:
+        real_store(principal, token)
+        if threading.current_thread().name == "first":
+            in_window.set()
+            second_finished.wait(WINDOW_SECONDS)
+
+    env.monkeypatch.setattr(auth_commands, "store_token", store_then_hold)
+
+    def body(command: Callable[[], None]) -> None:
+        name = threading.current_thread().name
+        try:
+            command()
+        except click.ClickException as exc:
+            refusals[name] = exc
+        finally:
+            if name == "second":
+                second_finished.set()
+
+    first_thread = threading.Thread(
+        target=body, args=(first,), name="first", daemon=True
+    )
+    second_thread = threading.Thread(
+        target=body, args=(second,), name="second", daemon=True
+    )
+    first_thread.start()
+    assert in_window.wait(
+        JOIN_SECONDS
+    ), "the first command never reached its keychain write"
+    second_thread.start()
+    first_thread.join(JOIN_SECONDS)
+    second_thread.join(JOIN_SECONDS)
+    assert not first_thread.is_alive() and not second_thread.is_alive()
+    return refusals
+
